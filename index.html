Provably Fair Crash Generation (server should generate, front simulates) function generateCrashPoint(){ // Betika uses: crash = 0.99 / (1 - r) where r = random 0-1 // 50% crash <2x, 15% crash >10x, 1% crash >100x let r = Math.random(); if(r < 0.02) return 1.00 + Math.random()*0.32; // instant crash 2% let crash = 0.99 / (1 - r); return Math.max(1.00, Math.min(crash, 1000)); // cap 1000x } // 2. Game State - 6. Game area with airplane + graph + voice let state = { phase: 'waiting', // waiting, flying, crashed multiplier: 1.00, crashPoint: 2.00, startTime: 0, bets: {} // {bet1: {amount, active}, bet2: {...}} }; // 3. Plane + Graph + Voice Animation (like Betika) const plane = document.getElementById('plane'); const multEl = document.getElementById('mult'); const trail = document.getElementById('trail'); // SVG const fill = document.getElementById('fill'); // SVG fill const status = document.getElementById('status'); function speakBetika(text){ // Betika voice: "Bet accepted", "Cash out", "Flew away" if(!voiceOn) return; speechSynthesis.speak(new SpeechSynthesisUtterance(text)); } function betikaLoop(timestamp){ if(state.phase === 'waiting'){ state.multiplier = 1.00; multEl.textContent = '1.00x'; plane.style.left = '4%'; plane.style.bottom = '20%'; plane.style.opacity='1'; trail.setAttribute('d','M0,85 L0,85'); status.style.display='block'; status.innerHTML='WAITING FOR NEXT ROUND
Place BET'; setTimeout(()=>{ state.crashPoint = generateCrashPoint(); state.phase = 'flying'; state.startTime = performance.now(); status.style.display='none'; speakBetika('Round started'); }, 3000); // Betika 3 sec between rounds } else if(state.phase === 'flying'){ let elapsed = (performance.now() - state.startTime) / 1000; // Betika curve: multiplier = 1.07 ^ elapsed (exponential) state.multiplier = Math.pow(1.0718, elapsed * 1.15); multEl.textContent = state.multiplier.toFixed(2)+'x'; if(state.multiplier > 2) multEl.style.color='#ffcc00'; if(state.multiplier > 5) multEl.style.color='#ff2b6a'; // Plane moves left->right with graph let progress = Math.min(elapsed / 15, 1); // 15 sec max flight let left = 4 + progress*76; let bottom = 20 + Math.pow(progress,1.5)*60; plane.style.left = left+'%'; plane.style.bottom = bottom+'%'; plane.style.transform = `rotate(${8+progress*25}deg)`; // Graph trail like Betika let path = `M0,85 Q${progress*45},${85-Math.pow(progress,1.2)*40} ${left},${100-bottom}`; trail.setAttribute('d', path); fill.setAttribute('d', path + ` L${left},100 L0,100 Z`); // Update cash out buttons Object.keys(state.bets).forEach(id=>{ if(state.bets[id].active){ let btn = document.getElementById('b'+id); btn.textContent = `CASH OUT KES ${(state.bets[id].amount * state.multiplier).toFixed(0)}`; } }); // Crash check if(state.multiplier >= state.crashPoint){ state.phase='crashed'; multEl.textContent = state.crashPoint.toFixed(2)+'x'; multEl.style.color='#ff0a54'; status.style.display='block'; status.innerHTML=`
FLEW AWAY!
${state.crashPoint.toFixed(2)}x`; plane.style.opacity='0'; speakBetika(`Flew away at ${state.crashPoint.toFixed(2)}`); // Lose un-cashed bets Object.keys(state.bets).forEach(id=>{ if(state.bets[id].active){ state.bets[id].active=false; addToHistory(state.bets[id].amount, 0, state.crashPoint, false); } }); // Add to top history bar (like Betika: 4.21x, 1.32x...) addCrashToBar(state.crashPoint); setTimeout(()=>{ state.phase='waiting'; }, 2500); } } requestAnimationFrame(betikaLoop); } // 4. BET + CASH OUT - like Betika function placeBet(id, amount){ if(state.bets[id]?.active){ // Cash out let win = state.bets[id].amount * state.multiplier; if(mode==='real') realBalance+=win; else demoBalance+=win; state.bets[id].active=false; speakBetika(`Cash out ${win.toFixed(0)}`); toast(`CASH OUT KES ${win.toFixed(0)} @${state.multiplier.toFixed(2)}x`); addToHistory(state.bets[id].amount, win, state.multiplier, true); return; } if(mode==='real' && amount > realBalance){ toast('Low REAL - Deposit to 0716580098'); return; } if(mode==='real') realBalance-=amount; else demoBalance-=amount; state.bets[id]={amount, active:true}; toast(`BET KES ${amount} placed`); } // Start requestAnimationFrame(betikaLoop);
